#=============================================================================
# File        : spi_master.sdc
# Project     : SPI_Master_Controller
# Tool        : Cadence Genus / Innovus  (also readable by any STA tool)
# Description : ASIC timing constraints for a 100 MHz implementation.
#
#   NOTE : the driving cell / library cell names below must be replaced by
#          cells that exist in the target standard-cell library.
#=============================================================================

#-----------------------------------------------------------------------------
# 0. Units
#-----------------------------------------------------------------------------
set_units -time ns -capacitance pF

#-----------------------------------------------------------------------------
# 1. Primary clock : 100 MHz
#-----------------------------------------------------------------------------
create_clock -name clk -period 10.000 -waveform {0.000 5.000} [get_ports clk]

# Clock uncertainty (jitter + skew budget)
set_clock_uncertainty -setup 0.150 [get_clocks clk]
set_clock_uncertainty -hold  0.050 [get_clocks clk]

# Clock transition (slew)
set_clock_transition -rise 0.100 [get_clocks clk]
set_clock_transition -fall 0.100 [get_clocks clk]

# Clock latency
set_clock_latency -source -max  1.000 [get_clocks clk]
set_clock_latency -source -min  0.500 [get_clocks clk]
set_clock_latency        -max   0.500 [get_clocks clk]
set_clock_latency        -min   0.200 [get_clocks clk]

#-----------------------------------------------------------------------------
# 2. Input / output drive & load
#-----------------------------------------------------------------------------
set_driving_cell -lib_cell BUFX2 -pin Y  [remove_from_collection [all_inputs] [get_ports clk]]
set_load        0.050                    [all_outputs]

set_max_transition 0.500 [current_design]
set_max_capacitance 0.200 [current_design]

#-----------------------------------------------------------------------------
# 3. Input delays
#    Reset / start are sampled by clk.
#-----------------------------------------------------------------------------
set_input_delay -clock clk -max 2.000 [get_ports {rst_n start}]
set_input_delay -clock clk -min 0.500 [get_ports {rst_n start}]

#-----------------------------------------------------------------------------
# 4. SPI interface
#    Fastest SCLK = 50 MHz (clk_divider = 1) -> 20 ns period.
#    A virtual clock represents the external SPI slave domain.
#-----------------------------------------------------------------------------
create_clock -name spi_sclk_virt -period 20.000

set_input_delay  -clock spi_sclk_virt -max 5.000 [get_ports spi_miso]
set_input_delay  -clock spi_sclk_virt -min 1.000 [get_ports spi_miso]

set_output_delay -clock spi_sclk_virt -max 5.000 \
                 [get_ports {spi_mosi spi_sclk spi_cs_n}]
set_output_delay -clock spi_sclk_virt -min 1.000 \
                 [get_ports {spi_mosi spi_sclk spi_cs_n}]

#-----------------------------------------------------------------------------
# 5. Configuration ports (quasi-static)
#    Uncomment when they are real chip inputs.
#-----------------------------------------------------------------------------
# set_input_delay -clock clk -max 3.000 [get_ports {tx_data[*] clk_divider[*] cpol cpha}]
# set_input_delay -clock clk -min 1.000 [get_ports {tx_data[*] clk_divider[*] cpol cpha}]

#-----------------------------------------------------------------------------
# 6. Exceptions
#-----------------------------------------------------------------------------
set_false_path -from [get_ports rst_n]
set_false_path -to   [get_ports spi_sclk]

#-----------------------------------------------------------------------------
# 7. Design rule / reporting helpers
#-----------------------------------------------------------------------------
set_max_fanout 32 [current_design]
